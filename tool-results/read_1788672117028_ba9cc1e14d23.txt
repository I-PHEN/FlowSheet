     1→/**
     2→ * Unit registry — the ammonia catalog: every unit type the flowsheet graph
     3→ * can be built from. Each type declares typed ports, spec fields with
     4→ * physical clamps/defaults, and a solve() that wraps the EXISTING physics
     5→ * in units.ts verbatim (same calls, same formatting — this is what lets
     6→ * the graph executor reproduce the legacy plant.ts numbers exactly).
     7→ *
     8→ * solve() receives inlet StreamStates and returns outlet StreamStates +
     9→ * the UnitResult payload (name/model/metrics/warnings). Wrappers read
    10→ * NOTHING outside their own inlets and specs — unit independence is what
    11→ * makes free composition safe.
    12→ */
    13→

    39→const C = (celsius: number) => celsius + 273.15;
    40→const BAR = (b: number) => b * 1e5;
    41→const MW = (kJh: number) => kJh / 3.6e6;
    42→const d1 = (x: number) => (Math.abs(x) >= 100 ? x.toFixed(0) : x.toFixed(1));
    43→const d2 = (x: number) => x.toFixed(2);
    44→
    45→export interface SolveCtx {
    46→  specs: Record<string, number | boolean | number[]>;
    47→  inlet: (port: string) => StreamState;
    48→  /** push a SECTION-level warning (mirrors the old fe.warnings list) */
    49→  warn: (msg: string) => void;
    50→}
    51→
    52→export interface SolveOut {
    53→  outlets: Record<string, StreamState>;
    54→  /** UnitResult minus id — the executor stamps the instance id */
    55→  result: Pick<UnitResult, 'name' | 'model' | 'metrics' | 'warnings'>;
    56→}
    57→
    58→export interface UnitTypeDef {
    59→  type: string;
    60→  /** default display name (instances may not override in v1) */
    61→  name: string;
    62→  model: (specs: Record<string, number | boolean | number[]>) => string;
    63→  ports: { in: PortDef[]; out: PortDef[] };
    64→  specFields: SpecField[];
    65→  solve: (ctx: SolveCtx) => SolveOut;
    66→  /**
    67→   * Present when this unit's outlet T and P are fully spec-determined
    68→   * (independent of its inlet). The executor tears recycle loops at such
    69→   * edges: the cut stream's T/P stay exact on every iteration.
    70→   */
    71→  fixedOutlet?: (specs: Record<string, number | boolean | number[]>) => { T: number; P: number };
    72→}
    73→
    74→const num = (s: Record<string, number | boolean | number[]>, k: string): number => s[k] as number;
    75→const arr = (s: Record<string, number | boolean | number[]>, k: string): number[] => s[k] as number[];

   806→export function resolveSpecs(unit: { type: string; specs: Record<string, number | boolean | number[]> }): Record<string, number | boolean | number[]> {
   807→  const def = getUnitType(unit.type);

   818→ * Initial tear guess for the synthesis loop: converter feed ≈ 5× make-up at
   819→ * a typical loop-gas composition (research Q9 tip). This is the same
   820→ * initializer the legacy plant.ts used — keeping it identical keeps the
   821→ * iteration path (and therefore the trace) identical.
   822→ */
   823→export function tearInit(makeup: Moles): Moles {
   824→  const tear0: Moles = zeroN();
   825→  const f0 = total(makeup) * 5;
   826→  const y0 = [0.58, 0.195, 0, 0, 0.11, 0.033, 0.025, 0, 0]; // H2 N2 CO CO2 CH4 AR NH3 H2O O2
   827→  for (let i = 0; i < N_SP; i++) tear0[i] = f0 * y0[i];
   828→  return tear0;
   829→}
