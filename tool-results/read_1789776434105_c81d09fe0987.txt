     4→     4→ * BuildCanvas — the stage of the builder studio.
     5→     5→ *
     6→     6→ * The graph arrives piecewise from the agent's tool events, so the layout is

     8→     8→ * bands. New units fade in as the engineer places them; streams are

    68→    68→function computeLayout(graph: FlowGraph): Layout {
    69→    69→  const units = graph.units;
    70→    70→  if (units.length === 0) return { width: 900, height: 360, units: new Map(), bandCount: 1 };
    71→    71→
    72→    72→  // adjacency (unit digraph, self-loops ignored)
    73→    73→  const preds = new Map<string, Set<string>>();
    74→    74→  const succs = new Map<string, Set<string>>();
    75→    75→  for (const u of units) {
    76→    76→    preds.set(u.id, new Set());
    77→    77→    succs.set(u.id, new Set());
    78→    78→  }
    79→    79→  for (const s of graph.streams) {
    80→    80→    if (s.to && s.to.unit !== s.from.unit) {
    81→    81→      preds.get(s.to.unit)?.add(s.from.unit);
    82→    82→      succs.get(s.from.unit)?.add(s.to.unit);
    83→    83→    }
    84→    84→  }
    85→    85→
    86→    86→  // BFS depth from units with no predecessors (feeds)
    87→    87→  const depth = new Map<string, number>();
    88→    88→  const queue: string[] = [];
    89→    89→  for (const u of units) {
    90→    90→    if ((preds.get(u.id) ?? new Set()).size === 0) {
    91→    91→      depth.set(u.id, 0);
    92→    92→      queue.push(u.id);
    93→    93→    }
    94→    94→  }
    95→    95→  while (queue.length > 0) {
    96→    96→    const cur = queue.shift()!;
    97→    97→    const d = depth.get(cur) ?? 0;
    98→    98→    for (const next of succs.get(cur) ?? []) {
    99→    99→      const nd = d + 1;
   100→   100→      if ((depth.get(next) ?? Infinity) > nd) {
   101→   101→        depth.set(next, nd);
   102→   102→        queue.push(next);
   103→   103→      }
   104→   104→    }
   105→   105→  }
   106→   106→  // unreachable units (shouldn't happen post-validation) park at depth 0
   107→   107→  for (const u of units) if (!depth.has(u.id)) depth.set(u.id, 0);
   108→   108→
   109→   109→  // group by depth, preserving insertion (build) order within a column
   110→   110→  const byDepth = new Map<number, string[]>();
   111→   111→  for (const u of units) {
   112→   112→    const d = depth.get(u.id) ?? 0;
   113→   113→    if (!byDepth.has(d)) byDepth.set(d, []);
   114→   114→    byDepth.get(d)!.push(u.id);
   115→   115→  }
   116→   116→  const maxDepth = Math.max(...byDepth.keys());
   117→   117→  const maxColLen = Math.max(...[...byDepth.values()].map((v) => v.length), 1);
   118→   118→
   119→   119→  const bandCount = Math.floor(maxDepth / BAND_COLS) + 1;
   120→   120→  const bandGap = 64;
   121→   121→  const placed = new Map<string, Placed>();
   122→   122→  for (const u of units) {
   123→   123→    const d = depth.get(u.id) ?? 0;
   124→   124→    const col = byDepth.get(d)!.indexOf(u.id);
   125→   125→    const band = Math.floor(d / BAND_COLS);
   126→   126→    // wrap the chain into bands: column position within the band, band rows
   127→   127→    // stacked vertically — keeps the canvas near-square instead of 20-wide
   128→   128→    const x = PAD + (d % BAND_COLS) * COL_W;
   129→   129→    const y = PAD + band * (maxColLen * ROW_H + bandGap) + col * ROW_H + (band % 2 === 1 ? 24 : 0);
   130→   130→    placed.set(u.id, { id: u.id, x, y, depth: d, col, band });
   131→   131→  }
   132→   132→
   133→   133→  const width = PAD * 2 + Math.min(maxDepth + 1, BAND_COLS) * COL_W;
   134→   134→  const height = PAD * 2 + bandCount * (maxColLen * ROW_H) + (bandCount - 1) * bandGap + 48;
   135→   135→  return { width: Math.max(width, 900), height: Math.max(height, 360), units: placed, bandCount };
   136→   136→}

   138→   138→function Arrow({ x, y, angle, color }: { x: number; y: number; angle: number; color: string }) {
   139→   139→  const L = 11;
   140→   140→  const W = 5.5;

   143→   143→      points={`${x},${y} ${x - L * Math.cos(angle) + W * Math.sin(angle)},${y - L * Math.sin(angle) - W * Math.cos(angle)} ${x - L * Math.cos(angle) - W * Math.sin(angle)},${y - L * Math.sin(angle) + W * Math.cos(angle)}`}

   146→   146→  );

   149→   149→const fmt = (x: number, d = 0) =>
   150→   150→  x.toLocaleString('en-US', { maximumFractionDigits: d, minimumFractionDigits: d });

   152→   152→/** keep a view inside the layout bounds (some overhang allowed, like the reference canvas) */
   153→   153→function clampView(v: View, L: Layout): View {
   154→   154→  const w = Math.min(L.width * 1.6, Math.max(L.width / 5, v.w));
   155→   155→  const h = w * (L.height / L.width);
   156→   156→  const x = Math.min(L.width - w * 0.25, Math.max(-w * 0.75, v.x));
   157→   157→  const y = Math.min(L.height - h * 0.25, Math.max(-h * 0.75, v.y));
   158→   158→  return { x, y, w, h };
   159→   159→}

   161→   161→/** live-value solves, memoized per graph snapshot (mid-build snapshots may be
   162→   162→    unsolvable — null is cached too, so nothing ever blocks or re-solves) */
   163→   163→const solveCache = new WeakMap<FlowGraph, PlantResult | null>();
   164→   164→
   165→   165→function solveFor(g: FlowGraph): PlantResult | null {
   166→   166→  if (!solveCache.has(g)) {
   167→   167→    let r: PlantResult | null = null;
   168→   168→    try {
   169→   169→      r = executeGraph(g);
   170→   170→    } catch {
   171→   171→      r = null; // mid-build snapshots may be unsolvable — that is fine
   172→   172→    }
   173→   173→    solveCache.set(g, r);
   174→   174→  }
   175→   175→  return solveCache.get(g) ?? null;
   176→   176→}

   178→   178→/** tooltip payload for a stream — live values when the graph solves, structure otherwise */
   179→   179→function streamTip(s: StreamEdge, result: PlantResult | null) {
   180→   180→  const rows: Array<{ k: string; v: string }> = [];
   181→   181→  const live = result?.streams[s.id];
   182→   182→  if (live) {
   183→   183→    const tot = live.n.reduce((a, b) => a + b, 0) || 1;
   184→   184→    const top = live.n
   185→   185→      .map((v, i) => ({ sp: SPECIES[i], x: v / tot }))
   186→   186→      .filter((e) => e.x > 0.005)
   187→   187→      .sort((a, b) => b.x - a.x)
   188→   188→      .slice(0, 4);
   189→   189→    rows.push(
   190→   190→      { k: 'T', v: `${fmt(live.T - 273.15)} °C` },
   191→   191→      { k: 'P', v: `${fmt(live.P / 1e5, 1)} bar` },
   192→   192→      { k: 'Flow', v: `${fmt(tot)} kmol/h` },
   193→   193→    );
   194→   194→    return {
   195→   195→      title: `${s.id} · ${s.name || s.cls}`,
   196→   196→      rows,
   197→   197→      foot: top.map((e) => `${e.sp} ${(e.x * 100).toFixed(1)}%`).join(' · '),
   198→   198→    };
   199→   199→  }
   200→   200→  rows.push(
   201→   201→    { k: 'from', v: s.from.unit },
   202→   202→    { k: 'to', v: s.to ? s.to.unit : 'sink' },
   203→   203→    { k: 'class', v: s.cls },
   204→   204→  );
   205→   205→  return { title: `${s.id} · ${s.name || s.cls}`, rows, foot: 'values appear once this snapshot solves' };
   206→   206→}

   407→   407→  /** letterbox-aware client→world mapping (view keeps the layout aspect) */
   408→   408→  const worldFromClient = useCallback((cx: number, cy: number) => {
